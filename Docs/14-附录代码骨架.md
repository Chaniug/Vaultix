## A. EncString 编解码（Bitwarden）

```kotlin
// core/crypto/bitwarden/EncString.kt
object EncString {
    private const val TYPE_AES_CBC_HMAC = 2

    fun parse(raw: String): Parsed {
        val (typeStr, payload) = raw.split(".", limit = 2)
        val type = typeStr.toInt()
        require(type == TYPE_AES_CBC_HMAC) { "不支持的加密类型: $type" }
        val parts = payload.split("|")
        require(parts.size == 3) { "密文格式错误" }
        return Parsed(
            iv = parts[0].b64(),
            ct = parts[1].b64(),
            mac = parts[2].b64(),
        )
    }

    fun decrypt(raw: String, key: SymmetricKey): SecureBytes {
        val p = parse(raw)
        // 1) 先验 MAC —— encrypt-then-MAC，绝不先解密
        val expected = HmacSha256.mac(key.macKey, p.iv + p.ct)
        require(MessageDigest.isEqual(expected, p.mac)) { "密文校验失败（已被篡改或密钥错误）" }
        // 2) 再解密
        return AesCbc.decrypt(key.encKey, p.iv, p.ct)   // 去 PKCS#7
    }

    fun encrypt(plain: SecureBytes, key: SymmetricKey): String {
        val iv = SecureRandom.nextBytes(16)
        val ct = AesCbc.encrypt(key.encKey, iv, plain)
        val mac = HmacSha256.mac(key.macKey, iv + ct)
        return listOf(
            TYPE_AES_CBC_HMAC.toString(),
            listOf(iv, ct, mac).joinToString("|") { it.b64() }
        ).joinToString(".")
    }
}

data class SymmetricKey(private val raw64: SecureBytes) {
    val encKey get() = raw64.view(0, 32)
    val macKey get() = raw64.view(32, 32)
    fun zero() = raw64.zero()
}
```

```kotlin
suspend fun deriveStretchedKey(
    password: SecureString, email: String, cfg: KdfConfig,
): StretchedKey = withContext(Dispatchers.Default) {
    val masterKey = when (cfg) {
        is KdfConfig.Pbkdf2 -> Pbkdf2.sha256(
            password.bytes, salt = email.trim().lowercase().toByteArray(), cfg.iterations, 32)
        is KdfConfig.Argon2id -> Argon2id.derive(
            password.bytes, salt = argonSaltOf(email), cfg.memoryKiB, cfg.iterations, cfg.parallelism, 32)
    }.also { password.clear() }

    ensureActive()   // 支持用户取消
    StretchedKey(
        enc = Hkdf.expand(masterKey, info = "enc".toByteArray(), 32),
        mac = Hkdf.expand(masterKey, info = "mac".toByteArray(), 32),
    ).also { masterKey.zero() }
}
```

## C. KDBX 4.x 解密管道

```kotlin
// data/kdbx/KdbxReader.kt
suspend fun open(stream: InputStream, credentials: KdbxCredentials): KdbxSession =
    withContext(Dispatchers.IO) {
        val header = KdbxHeader.read(stream)                 // 签名、版本、字段
        require(header.major == 4) { throw UnsupportedFormat(header.version) }   // 3.1 及更早直接拒
        require(header.kdfParams.uuid == ARGON2ID) { throw UnsupportedFormat("KDF") }
        require(header.cipherId == AES_CBC || header.cipherId == CHACHA20) { throw UnsupportedFormat("Cipher") }

        val composite = CompositeKey.of(credentials)         // SHA256(密码‖KeyFile‖挑战响应)
        val transformed = Argon2id.transform(composite, header.kdfParams)

        val masterKey  = Sha256.hash(header.masterSeed + transformed)
        val hmacBase   = Sha512.hash(header.masterSeed + transformed + 0x01)
        // header 校验：i = 0xFFFFFFFF_FFFFFFFF
        verifyHeaderHmac(stream, header, hmacKeyOf(hmacBase, index = -1L))

        val payload = readHmacBlocks(stream, hmacBase)       // 逐块校验后拼接
        val xml = Decryptor(header.cipherId, masterKey, header.iv)
            .decrypt(payload).let { if (header.compressed) Gzip.unzip(it) else it }

        val inner = InnerHeader.read(xml)                    // 保护流 ID/Key、二进制表
        require(inner.streamId == CHACHA20) { throw UnsupportedFormat("InnerStream") }
        KdbxSession(meta, root, entries, inner.streamKey, ...)
    }

private fun readHmacBlocks(stream: InputStream, hmacBase: SecureBytes): SecureBytes {
    var index = 0L
    val out = ByteArrayOutputStream()
    while (true) {
        val mac  = stream.readNBytes(32)
        val len  = stream.readNBytes(4).toIntLe()
        if (len == 0) break                                  // 终止块
        val data = stream.readNBytes(len)
        val expected = HmacSha256.mac(hmacKeyOf(hmacBase, index), data)
        if (!MessageDigest.isEqual(mac, expected)) throw KdbxCorrupted("block#$index")
        out.write(data); index++
    }
    return SecureBytes(out.toByteArray())
}
```

```kotlin
// 仅支持 ChaCha20；打开文件时 InnerRandomStreamID ≠ 3 已在读取阶段拒绝
class ProtectedStream(private val streamKey: SecureBytes) {
    private val key = Sha256.hash(streamKey)                      // 32B
    private val nonce = CHACHA20_NONCE_12B                        // E830094B97205D2A + 4B 零

    fun decrypt(b64: String): SecureBytes = ChaCha20.crypt(key, nonce, b64.b64())
    fun encrypt(plain: SecureBytes): String = ChaCha20.crypt(key, nonce, plain).b64()
}
```

## F. 生物识别解锁

```kotlin
class BiometricUnlocker @Inject constructor(
    private val keystore: KeystoreKeyProvider,
    private val store: SecureCredentialStore,
) {
    suspend fun enable(vaultId: String, masterSecret: SecureBytes) {
        val sealed = keystore.encrypt(alias = "vaultix_biometric", plain = masterSecret)
        store.put("biometric_secret_$vaultId", sealed)
    }

    suspend fun unlock(vaultId: String, cryptoObject: BiometricPrompt.CryptoObject): SecureBytes {
        val sealed = store.get("biometric_secret_$vaultId") ?: throw BiometricInvalidated
        return keystore.decrypt("vaultix_biometric", sealed, cryptoObject)
            // 需捕获 KeyPermanentlyInvalidatedException → BiometricInvalidated
    }
}
```

```kotlin
class VaultixAutofillService : AutofillService() {
    override fun onFillRequest(req: FillRequest, signal: CancellationSignal, cb: FillCallback) {
        val ctx = AssistParser.parse(req.fillContexts.last())      // 包名 + 域名 + 字段

        }.build()
```

```kotlin
class VaultHomeViewModel @Inject constructor(
```
